function carregarOrdemColunas() {

  try {

    const saved = localStorage.getItem(COL_ORDER_KEY);

    if (saved) {

      const arr = JSON.parse(saved);

      // valida que todas as chaves existem

      const keys = COLUNAS_DEF.map(c => c.key);

      if (arr.every(k => keys.includes(k)) && arr.length === keys.length) return arr;

    }

  } catch(e) {}

  return COLUNAS_DEF.map(c => c.key);

}

// ── Formatação ────────────────────────────────────────────────

function fmtData(d) {

  if (!d) return '-';

  const [y,m,dia] = d.split('-');

  return `${dia}/${m}/${y}`;

}

function fmtValor(v) {

  return 'R$ ' + Number(v).toLocaleString('pt-BR', { minimumFractionDigits:2, maximumFractionDigits:2 });

}

function badgeClass(s) {

  if (s === 'NÃO ACATADA') return 'badge-nao';

  if (s === 'ACATADA') return 'badge-ok';

  return 'badge-pend';

}

function parseValor(str) {

  return parseFloat(String(str).replace(/\./g,'').replace(',','.')) || 0;

}

function celula(col, n) {

  switch(col.key) {

    case 'data':       return fmtData(n.data);

    case 'fornecedor': return `<span title="${n.fornecedor}">${n.fornecedor}</span>`;

    case 'nf':         return n.nf;

    case 'valor':      return `<span style="font-weight:600">${fmtValor(n.valor)}</span>`;

    case 'chegada':    return n.chegada || '-';

    case 'temp':       return n.temp || '-';

    case 'saida':      return n.saida || '-';

    case 'conferente': return `<span title="${n.conferente||'-'}">${n.conferente||'-'}</span>`;

    case 'prevencao':  return `<span title="${n.prevencao||'-'}">${n.prevencao||'-'}</span>`;

    case 'obs':        return `<span title="${n.obs||'-'}">${n.obs||'-'}</span>`;

    case 'status':     return `<span class="badge ${badgeClass(n.status)}">${n.status}</span>`;

    case 'acoes':      return `<div class="row-actions">

      <button class="icon-btn" title="Editar" onclick="editarNota(${n.id})">✏️</button>

      <button class="icon-btn del" title="Excluir" onclick="excluirNota(${n.id})">🗑️</button>

    </div>`;

    default: return '-';

  }

}

  const tr = document.querySelector('#thead-row tr');

  tr.innerHTML = '';

  colunasOrdenadas().forEach((col, idx) => {

    const th = document.createElement('th');

    th.dataset.idx = idx;

    th.dataset.key = col.key;

    if (col.key === 'acoes') {

      th.className = 'actions-col';

      th.style.width = '70px';

      th.innerHTML = `<div class="th-inner">Ações</div>`;

    } else {

      if (col.sortable) th.classList.add('sortable');

      if (col.key === sortCol) th.classList.add('sorted');

      th.innerHTML = `<div class="th-inner">

        <span class="drag-handle" draggable="true" title="Arrastar para mover coluna">☰</span>

        <span ${col.sortable ? `onclick="ordenar('${col.key}')" style="cursor:pointer"` : ''}>${col.label}${col.sortable ? ' <span style="opacity:.5;font-size:10px">↕</span>' : ''}</span>

      </div>`;

      // drag events no handle

      const handle = th.querySelector('.drag-handle');

      handle.addEventListener('dragstart', e => {

        dragSrcIdx = idx;

        th.classList.add('dragging');

        e.dataTransfer.effectAllowed = 'move';

      });

      handle.addEventListener('dragend', () => {

        document.querySelectorAll('thead th').forEach(t => {

          t.classList.remove('dragging','drag-over');

        });

        dragSrcIdx = null;

      });

      th.addEventListener('dragover', e => {

        if (dragSrcIdx === null || dragSrcIdx === idx) return;

        const tgt = colunasOrdenadas()[idx];

        if (tgt && tgt.fixed) return;

        e.preventDefault();

        e.dataTransfer.dropEffect = 'move';

        document.querySelectorAll('thead th').forEach(t => t.classList.remove('drag-over'));

        th.classList.add('drag-over');

      });

      th.addEventListener('drop', e => {

        if (dragSrcIdx === null || dragSrcIdx === idx) return;

        const tgt = colunasOrdenadas()[idx];

        if (tgt && tgt.fixed) return;

        e.preventDefault();

        const moved = colOrder.splice(dragSrcIdx, 1)[0];

        colOrder.splice(idx, 0, moved);

        salvarOrdemColunas();

        renderCabecalho();

        renderTabela();

        toast('Coluna movida!');

      });

    }

    tr.appendChild(th);

  });

}


// ── Filtros ───────────────────────────────────────────────────

function filtradas() {

  const forn  = document.getElementById('f-fornecedor').value.toLowerCase();

  const nf    = document.getElementById('f-nf').value.toLowerCase();

  const conf  = document.getElementById('f-conferente').value.toLowerCase();

  const st    = document.getElementById('f-status').value;

  const di    = document.getElementById('f-data-ini').value;

  const df    = document.getElementById('f-data-fim').value;

  return notas.filter(n => {

    if (forn && !n.fornecedor.toLowerCase().includes(forn)) return false;

    if (nf   && !n.nf.toLowerCase().includes(nf)) return false;

    if (conf && !(n.conferente||'').toLowerCase().includes(conf)) return false;

    if (st   && n.status !== st) return false;

    if (di   && n.data < di) return false;

    if (df   && n.data > df) return false;

    return true;

  });

}

  filtrosVisiveis = !filtrosVisiveis;

  document.getElementById('filtros-row').style.display = filtrosVisiveis ? 'flex' : 'none';

}

function limparFiltros() {

  ['f-fornecedor','f-nf','f-conferente','f-status','f-data-ini','f-data-fim'].forEach(id => {

    document.getElementById(id).value = '';

  });

  renderTabela();

}


// ── Ordenação ─────────────────────────────────────────────────

function ordenar(col) {

  if (sortCol === col) sortAsc = !sortAsc;

  else { sortCol = col; sortAsc = true; }

  renderCabecalho();

  renderTabela();

}


// ── Render ────────────────────────────────────────────────────

function renderTabela() {

  const lista = filtradas().sort((a, b) => {

    let va = a[sortCol], vb = b[sortCol];

    if (sortCol === 'valor') { va = +va; vb = +vb; }

    if (va < vb) return sortAsc ? -1 : 1;

    if (va > vb) return sortAsc ? 1 : -1;

    return 0;

  });


  document.getElementById('s-total').textContent = notas.length;

  document.getElementById('s-valor').textContent = fmtValor(notas.reduce((s,n) => s + Number(n.valor), 0));

  document.getElementById('s-nao').textContent = notas.filter(n => n.status === 'NÃO ACATADA').length;

  document.getElementById('s-ok').textContent  = notas.filter(n => n.status === 'ACATADA').length;

  document.getElementById('count-label').textContent = `${lista.length} nota${lista.length !== 1 ? 's' : ''} encontrada${lista.length !== 1 ? 's' : ''}`;


  const tbody = document.getElementById('tabela-body');

  const empty = document.getElementById('empty-msg');

  if (lista.length === 0) { tbody.innerHTML = ''; empty.style.display = 'block'; return; }

  empty.style.display = 'none';


  const cols = colunasOrdenadas();

  tbody.innerHTML = lista.map(n =>

    `<tr>${cols.map(col => `<td>${celula(col, n)}</td>`).join('')}</tr>`

  ).join('');

}

function salvarNota() {

  const fornecedor = document.getElementById('m-fornecedor').value.trim();

  const nf   = document.getElementById('m-nf').value.trim();

  const data = document.getElementById('m-data').value;

  if (!fornecedor) { toast('Informe o fornecedor.'); return; }

  if (!nf)         { toast('Informe o número da NF.'); return; }

  if (!data)       { toast('Informe a data.'); return; }


  const nota = {

    data, fornecedor, nf,

    valor:      parseValor(document.getElementById('m-valor').value),

    chegada:    document.getElementById('m-chegada').value   || '-',

    temp:       document.getElementById('m-temp').value.trim()  || '-',

    saida:      document.getElementById('m-saida').value     || '-',

    conferente: document.getElementById('m-conferente').value.trim() || '-',

    prevencao:  document.getElementById('m-prevencao').value.trim()  || '-',

    obs:        document.getElementById('m-obs').value.trim()  || '-',

    status:     document.getElementById('m-status').value,

  };


  if (editId) {

    const idx = notas.findIndex(n => n.id === editId);

    if (idx !== -1) notas[idx] = { ...notas[idx], ...nota };

    toast('Nota atualizada!');

  } else {

    notas.unshift({ id: nextId++, ...nota });

    toast('Nota adicionada!');

  }

  salvarStorage();

  fecharModal();

  renderTabela();

}

}

}


// ── Exportar ──────────────────────────────────────────────────

function exportarCSV() {

  const cols = colunasOrdenadas().filter(c => c.key !== 'acoes');

  const header = cols.map(c => c.label).join(';');

  const rows = filtradas().map(n => cols.map(col => {

    let v;

    switch(col.key) {

      case 'data':   v = fmtData(n.data); break;

      case 'valor':  v = fmtValor(n.valor); break;

      default:       v = n[col.key] || '-';

    }

    return `"${String(v).replace(/"/g,'""')}"`;

  }).join(';'));

  const csv = [header, ...rows].join('\r\n');

  const blob = new Blob(['\uFEFF'+csv], { type: 'text/csv;charset=utf-8' });

  const url = URL.createObjectURL(blob);

  const a = document.createElement('a');

  a.href = url;

  a.download = `notas_fiscais_${new Date().toISOString().slice(0,10)}.csv`;

  document.body.appendChild(a); a.click();

  document.body.removeChild(a);

  URL.revokeObjectURL(url);

  toast('CSV exportado!');

}

  clearTimeout(toastTimer);

  toastTimer = setTimeout(() => el.classList.remove('show'), 2500);

}
